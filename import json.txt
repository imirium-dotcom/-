class ModelCoordinator:
    def __init__(
        self,
        key_manager: Optional[KeyManager] = None,
        ollama_url: str = OLLAMA_URL,
        timeout: int = REQUEST_TIMEOUT,
    ):
        self.key_manager = key_manager or KeyManager()
        self.ollama_url = ollama_url.rstrip("/")
        self.chat_url = f"{self.ollama_url}/api/chat"
        self.tags_url = f"{self.ollama_url}/api/tags"
        self.show_url = f"{self.ollama_url}/api/show"
        self.timeout = timeout
        self.last_usage = {}
        self.last_errors = {}

    # ---------------------------------------------------------
    # PROVIDER / KEY MANAGER
    # ---------------------------------------------------------

    def _get_provider_key(self, provider: str) -> Optional[str]:
        provider = provider.strip().lower()

        try:
            key = self.key_manager.get_key(provider)
            if key:
                return key
        except Exception:
            pass

        env_names = {
            "openai": ["OPENAI_API_KEY"],
            "gemini": ["GEMINI_API_KEY", "GOOGLE_API_KEY"],
            "anthropic": ["ANTHROPIC_API_KEY"],
            "openrouter": ["OPENROUTER_API_KEY"],
            "groq": ["GROQ_API_KEY"],
            "mistral": ["MISTRAL_API_KEY"],
            "deepseek": ["DEEPSEEK_API_KEY"],
            "cohere": ["COHERE_API_KEY"],
            "huggingface": ["HF_TOKEN", "HUGGINGFACEHUB_API_TOKEN"],
            "cloudflare": ["CLOUDFLARE_API_TOKEN"],
            "nvidia": ["NVIDIA_API_KEY"],
            "replicate": ["REPLICATE_API_TOKEN"],
            "together": ["TOGETHER_API_KEY"],
            "fireworks": ["FIREWORKS_API_KEY"],
            "perplexity": ["PERPLEXITY_API_KEY"],
        }

        for env_name in env_names.get(provider, []):
            value = os.getenv(env_name)
            if value:
                return value

        return None

    def _provider_has_key(self, provider: str) -> bool:
        return bool(self._get_provider_key(provider))

    def _get_provider_config(self, provider: str) -> dict:
        config = {}

        try:
            if hasattr(self.key_manager, "list_providers"):
                providers = self.key_manager.list_providers()

                if isinstance(providers, dict):
                    item = providers.get(provider, {})
                    if isinstance(item, dict):
                        config.update(item)

                elif isinstance(providers, list):
                    for item in providers:
                        if isinstance(item, dict):
                            name = str(
                                item.get("provider")
                                or item.get("name")
                                or item.get("id")
                                or ""
                            ).lower()

                            if name == provider.lower():
                                config.update(item)
                                break
        except Exception:
            pass

        return config

    def list_models(self) -> dict:
        try:
            response = requests.get(
                self.tags_url,
                timeout=10,
            )
            response.raise_for_status()
            payload = response.json()

            models = []

            for item in payload.get("models", []):
                name = item.get("name") or item.get("model")

                if name:
                    models.append(name)

            return {
                "available": True,
                "models": models,
                "error": None,
            }

        except Exception as exc:
            self.last_errors["ollama"] = str(exc)

            return {
                "available": False,
                "models": [],
                "error": str(exc),
            }

    def is_model_available(self, model_name: str) -> bool:
        result = self.list_models()

        if not result["available"]:
            return False

        models = result["models"]

        if model_name in models:
            return True

        requested_base = model_name.split(":")[0]

        return any(
            model.split(":")[0] == requested_base
            for model in models
        )

    def get_available_model_names(self) -> list:
        available = self.list_models()
        result = list(available["models"])

        if self._provider_has_key("openai"):
            result.append(f"openai:{OPENAI_DEFAULT_MODEL}")

        try:
            catalog = self.key_manager.list_catalog()

            if isinstance(catalog, dict):
                for provider_name, provider_data in catalog.items():
                    if not isinstance(provider_data, dict):
                        continue

                    if not provider_data.get("enabled", True):
                        continue

                    if not self._provider_has_key(provider_name):
                        continue

                    models = provider_data.get("models", [])

                    if isinstance(models, list):
                        for model in models:
                            if isinstance(model, str):
                                result.append(
                                    f"{provider_name}:{model}"
                                )
                            elif isinstance(model, dict):
                                model_name = (
                                    model.get("id")
                                    or model.get("name")
                                )

                                if model_name:
                                    result.append(
                                        f"{provider_name}:{model_name}"
                                    )
        except Exception:
            pass

        return list(dict.fromkeys(result))

    def get_model_details(self, model_name: str) -> dict:
        try:
            response = requests.post(
                self.show_url,
                json={"name": model_name},
                timeout=20,
            )
            response.raise_for_status()
            return response.json()

        except Exception as exc:
            return {
                "error": str(exc),
                "model": model_name,
            }

    def select_models(self, task_type: str) -> list:
        available = self.get_available_model_names()
        preferred = self.preferred_models(task_type)

        selected = [
            model
            for model in preferred
            if model in available
        ]

        if not selected:
            selected = [
                model
                for model in available
                if not model.startswith("openai:")
            ]

        if not selected:
            selected = [
                model
                for model in available
                if ":" in model
            ]

        if not selected:
            selected = [
                model
                for model in available
                if model.startswith("openai:")
            ]

        return list(dict.fromkeys(selected))

    def build_system_prompt(
        self,
        task_type: str,
        extra_instructions: Optional[str] = None,
    ) -> str:
        parts = [
            ARMITAGE_CONTEXT,
            f"Тип задачи: {task_type}.",

            "Не придумывай результаты выполнения инструментов.",
        ]

        if extra_instructions:
            parts.append(str(extra_instructions))

        return "\n\n".join(parts)

    def clean_context(self, context) -> list:
        if context is None:
            return []

        if isinstance(context, str):
            return [
                {
                    "role": "user",
                    "content": context,
                }
            ]

        if not isinstance(context, list):
            return []

        cleaned = []

        for item in context:
            if not isinstance(item, dict):
                continue

            role = item.get("role")
            content = item.get("content")

            if role not in ("system", "user", "assistant", "developer"):
                continue

            if content is None:
                continue

            if isinstance(content, str):
                cleaned.append(
                    {
                        "role": role,
                        "content": content,
                    }
                )
            elif isinstance(content, list):
                cleaned.append(
                    {
                        "role": role,
                        "content": content,
                    }
                )

        return cleaned

    def _request_openai_model(
        self,
        model: str,
        messages: list,
        system_prompt: str,
        max_tokens: int,
    ) -> dict:
        api_key = self._get_provider_key("openai")

        if not api_key:
            return {
                "success": False,
                "complete": False,
                "model": model,
                "text": "",
                "error": "Ключ OpenAI не найден в KeyManager или переменных окружения.",
            }

        requested_model = model

        if requested_model.startswith("openai:"):
            requested_model = requested_model.split(":", 1)[1]

        if not requested_model:
            requested_model = OPENAI_DEFAULT_MODEL

        input_messages = []

        for message in messages:
            role = message.get("role", "user")

            if role == "system":
                continue

            if role == "developer":
                role = "developer"

            content = message.get("content", "")

            if isinstance(content, list):
                input_messages.append(
                    {
                        "role": role,
                        "content": content,
                    }
                )
            else:
                input_messages.append(
                    {
                        "role": role,
                        "content": str(content),
                    }
                )

        payload = {
            "model": requested_model,
            "instructions": system_prompt,
            "input": input_messages,
            "max_output_tokens": max_tokens,
        }

        headers = {
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json",
        }

        started = time.time()

        try:
            response = requests.post(
                OPENAI_API_URL,
                headers=headers,
                json=payload,
                timeout=self.timeout,
            )

            elapsed = round(time.time() - started, 2)

            if not response.ok:
                try:
                    error_data = response.json()
                except Exception:
                    error_data = response.text

                error_message = (
                    error_data.get("error", {}).get("message", str(error_data))
                    if isinstance(error_data, dict)
                    else str(error_data)
                )

                self.last_errors["openai"] = error_message

                return {
                    "success": False,
                    "complete": False,
                    "model": f"openai:{requested_model}",
                    "text": "",
                    "error": error_message,
                    "status_code": response.status_code,
                    "elapsed": elapsed,
                }

            data = response.json()
            output_text = data.get("output_text", "")

            if not output_text:
                chunks = []

                for item in data.get("output", []):
                    for content_item in item.get("content", []):
                        if content_item.get("type") in (
                            "output_text",
                            "text",
                        ):
                            text_value = content_item.get("text", "")

                            if text_value:
                                chunks.append(text_value)

                output_text = "\n".join(chunks)

            usage = data.get("usage", {})

            self.last_usage[f"openai:{requested_model}"] = usage

            status = data.get("status", "")
            incomplete_details = data.get("incomplete_details")

            complete = (
                status != "incomplete"
                and not incomplete_details
                and bool(output_text.strip())
            )

            return {
                "success": bool(output_text.strip()),
                "complete": complete,
                "model": f"openai:{requested_model}",
                "text": output_text.strip(),
                "error": None if output_text.strip() else (
                    "OpenAI вернул пустой ответ."
                ),
                "elapsed": elapsed,
                "usage": usage,
                "status": status,
                "incomplete_details": incomplete_details,
            }

        except Exception as exc:
            self.last_errors["openai"] = str(exc)

            return {
                "success": False,
                "complete": False,
                "model": f"openai:{requested_model}",
                "text": "",
                "error": str(exc),
            }

    def _request_openai_compatible_model(
        self,
        provider: str,
        model: str,
        messages: list,
        system_prompt: str,
        max_tokens: int,
    ) -> dict:
        api_key = self._get_provider_key(provider)

        if not api_key:
            return {
                "success": False,
                "complete": False,
                "model": f"{provider}:{model}",
                "text": "",
                "error": f"Ключ провайдера {provider} не найден.",
            }

        config = self._get_openai_compatible_config(provider)

        requested_model = model

        if ":" in requested_model:
            prefix, remainder = requested_model.split(":", 1)

            if prefix.lower() == provider.lower():
                requested_model = remainder

        if not requested_model:
            requested_model = config.get("default_model", "")

        if not requested_model:
            return {
                "success": False,
                "complete": False,
                "model": f"{provider}:{model}",
                "text": "",
                "error": f"Не указана модель для {provider}.",
            }

        base_url = config.get("base_url")

        if not base_url:
            return {
                "success": False,
                "complete": False,
                "model": f"{provider}:{requested_model}",
                "text": "",
                "error": f"Не задан API URL для {provider}.",
            }

        compatible_messages = [
            {
                "role": "system",
                "content": system_prompt,
            }
        ]

        for message in messages:
            role = message.get("role", "user")

            if role not in ("user", "assistant", "system"):
                continue

            content = message.get("content", "")

            compatible_messages.append(
                {
                    "role": role,
                    "content": content,
                }
            )

        payload = {
            "model": requested_model,
            "messages": compatible_messages,
            "max_tokens": max_tokens,
            "temperature": 0.3,
        }

        headers = {
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json",
        }

        if provider == "openrouter":
            headers["HTTP-Referer"] = "http://localhost"
            headers["X-Title"] = "Armitage AI Core"

        started = time.time()

        try:
            response = requests.post(
                f"{base_url}/chat/completions",
                headers=headers,
                json=payload,
                timeout=self.timeout,
            )

            elapsed = round(time.time() - started, 2)

            if not response.ok:
                try:
                    error_data = response.json()
                except Exception:
                    error_data = response.text

                if isinstance(error_data, dict):
                    error_message = (
                        error_data.get("error", {}).get("message")
                        if isinstance(error_data.get("error"), dict)
                        else str(error_data.get("error", error_data))
                    )
                else:
                    error_message = str(error_data)

                self.last_errors[provider] = error_message

                return {
                    "success": False,
                    "complete": False,
                    "model": f"{provider}:{requested_model}",
                    "text": "",
                    "error": error_message,
                    "status_code": response.status_code,
                    "elapsed": elapsed,
                }

            data = response.json()
            choices = data.get("choices", [])

            output_text = ""

            if choices:
                message_data = choices[0].get("message", {})
                content = message_data.get("content", "")

                if isinstance(content, str):
                    output_text = content
                elif isinstance(content, list):
                    output_text = "\n".join(
                        str(item.get("text", ""))
                        for item in content
                        if isinstance(item, dict)
                    )

            usage = data.get("usage", {})

            self.last_usage[f"{provider}:{requested_model}"] = usage

            finish_reason = (
                choices[0].get("finish_reason")
                if choices
                else None
            )

            complete = (
                bool(output_text.strip())
                and finish_reason not in ("length", "content_filter")
            )

            return {
                "success": bool(output_text.strip()),
                "complete": complete,
                "model": f"{provider}:{requested_model}",
                "text": output_text.strip(),
                "error": None if output_text.strip() else (
                    "Провайдер вернул пустой ответ."
                ),
                "elapsed": elapsed,
                "usage": usage,
                "finish_reason": finish_reason,
            }

        except Exception as exc:
            self.last_errors[provider] = str(exc)

            return {
                "success": False,
                "complete": False,
                "model": f"{provider}:{requested_model}",
                "text": "",
                "error": str(exc),
            }

    def _request_gemini_model(
        self,
        model: str,
        messages: list,
        system_prompt: str,
        max_tokens: int,
    ) -> dict:
        api_key = self._get_provider_key("gemini")

        if not api_key:
            return {
                "success": False,
                "complete": False,
                "model": f"gemini:{model}",
                "text": "",
                "error": "Ключ Gemini не найден.",
            }

        requested_model = model

        if requested_model.startswith("gemini:"):
            requested_model = requested_model.split(":", 1)[1]

        if not requested_model:
            requested_model = "gemini-2.5-flash"

        contents = []

        for message in messages:
            role = message.get("role", "user")

            if role == "assistant":
                role = "model"
            elif role not in ("user", "model"):
                continue

            content = message.get("content", "")

            if not isinstance(content, str):
                content = str(content)

            contents.append(
                {
                    "role": role,
                    "parts": [{"text": content}],
                }
            )

        payload = {
            "systemInstruction": {
                "parts": [{"text": system_prompt}],
            },
            "contents": contents,
            "generationConfig": {
                "maxOutputTokens": max_tokens,
                "temperature": 0.3,
            },
        }

        url = (
            "https://generativelanguage.googleapis.com/v1beta/models/"
            f"{requested_model}:generateContent"
        )

        started = time.time()

        try:
            response = requests.post(
                url,
                headers={
                    "x-goog-api-key": api_key,
                    "Content-Type": "application/json",
                },
                json=payload,
                timeout=self.timeout,
            )

            elapsed = round(time.time() - started, 2)

            if not response.ok:
                try:
                    error_data = response.json()
                except Exception:
                    error_data = response.text

                error_message = (
                    error_data.get("error", {}).get("message", str(error_data))
                    if isinstance(error_data, dict)
                    else str(error_data)
                )

                self.last_errors["gemini"] = error_message

                return {
                    "success": False,
                    "complete": False,
                    "model": f"gemini:{requested_model}",
                    "text": "",
                    "error": error_message,
                    "status_code": response.status_code,
                    "elapsed": elapsed,
                }

            data = response.json()
            candidates = data.get("candidates", [])
            output_text = ""

            if candidates:
                parts = (
                    candidates[0]
                    .get("content", {})
                    .get("parts", [])
                )

                output_text = "\n".join(
                    part.get("text", "")
                    for part in parts
                    if isinstance(part, dict)
                )

            usage = data.get("usageMetadata", {})
            self.last_usage[f"gemini:{requested_model}"] = usage

            finish_reason = (
                candidates[0].get("finishReason")
                if candidates
                else None
            )

            return {
                "success": bool(output_text.strip()),
                "complete": (
                    bool(output_text.strip())
                    and finish_reason not in (
                        "MAX_TOKENS",
                        "SAFETY",
                        "RECITATION",
                    )
                ),
                "model": f"gemini:{requested_model}",
                "text": output_text.strip(),
                "error": None if output_text.strip() else (
                    "Gemini вернул пустой ответ."
                ),
                "elapsed": elapsed,
                "usage": usage,
                "finish_reason": finish_reason,
            }

        except Exception as exc:
            self.last_errors["gemini"] = str(exc)

            return {
                "success": False,
                "complete": False,
                "model": f"gemini:{requested_model}",
                "text": "",
                "error": str(exc),
            }

    def _request_anthropic_model(
        self,
        model: str,
        messages: list,
        system_prompt: str,
        max_tokens: int,
    ) -> dict:
        api_key = self._get_provider_key("anthropic")

        if not api_key:
            return {
                "success": False,
                "complete": False,
                "model": f"anthropic:{model}",
                "text": "",
                "error": "Ключ Anthropic не найден.",
            }

        requested_model = model

        if requested_model.startswith("anthropic:"):
            requested_model = requested_model.split(":", 1)[1]

        if not requested_model:
            requested_model = "claude-3-5-haiku-latest"

        anthropic_messages = []

        for message in messages:
            role = message.get("role", "user")

            if role not in ("user", "assistant"):
                continue

            content = message.get("content", "")

            anthropic_messages.append(
                {
                    "role": role,
                    "content": str(content),
                }
            )

        payload = {
            "model": requested_model,
            "system": system_prompt,
            "messages": anthropic_messages,
            "max_tokens": max_tokens,
        }

        started = time.time()

        try:
            response = requests.post(
                "https://api.anthropic.com/v1/messages",
                headers={
                    "x-api-key": api_key,
                    "anthropic-version": "2023-06-01",
                    "Content-Type": "application/json",
                },
                json=payload,
                timeout=self.timeout,
            )

            elapsed = round(time.time() - started, 2)

            if not response.ok:
                try:
                    error_data = response.json()
                except Exception:
                    error_data = response.text

                error_message = (
                    error_data.get("error", {}).get("message", str(error_data))
                    if isinstance(error_data, dict)
                    else str(error_data)
                )

                self.last_errors["anthropic"] = error_message

                return {
                    "success": False,
                    "complete": False,
                    "model": f"anthropic:{requested_model}",
                    "text": "",
                    "error": error_message,
                    "status_code": response.status_code,
                    "elapsed": elapsed,
                }

            data = response.json()

            output_text = "\n".join(
                item.get("text", "")
                for item in data.get("content", [])
                if isinstance(item, dict)
                and item.get("type") == "text"
            )

            usage = data.get("usage", {})
            self.last_usage[f"anthropic:{requested_model}"] = usage

            stop_reason = data.get("stop_reason")

            return {
                "success": bool(output_text.strip()),
                "complete": (
                    bool(output_text.strip())
                    and stop_reason != "max_tokens"
                ),
                "model": f"anthropic:{requested_model}",
                "text": output_text.strip(),
                "error": None if output_text.strip() else (
                    "Anthropic вернул пустой ответ."
                ),
                "elapsed": elapsed,
                "usage": usage,
                "stop_reason": stop_reason,
            }

        except Exception as exc:
            self.last_errors["anthropic"] = str(exc)

            return {
                "success": False,
                "complete": False,
                "model": f"anthropic:{requested_model}",
                "text": "",
                "error": str(exc),
            }

    # ---------------------------------------------------------
    # MODEL REQUEST ROUTER
    # ---------------------------------------------------------

    def _request_model(
        self,
        model: str,
        messages: list,
        system_prompt: str,
        max_tokens: int,
    ) -> dict:
        if model.startswith("openai:"):
            return self._request_openai_model(
                model,
                messages,
                system_prompt,
                max_tokens,
            )

        if ":" in model:
            provider, model_name = model.split(":", 1)
            provider = provider.lower()

            if provider == "gemini":
                return self._request_gemini_model(
                    model,
                    messages,
                    system_prompt,
                    max_tokens,
                )

            if provider == "anthropic":
                return self._request_anthropic_model(
                    model,
                    messages,
                    system_prompt,
                    max_tokens,
                )

            compatible_providers = {
                "openrouter",
                "groq",
                "mistral",
                "deepseek",
                "together",
                "fireworks",
                "nvidia",
                "perplexity",
            }

            if provider in compatible_providers:
                return self._request_openai_compatible_model(
                    provider,
                    model_name,
                    messages,
                    system_prompt,
                    max_tokens,
                )

        return self._request_ollama_model(
            model,
            messages,
            system_prompt,
            max_tokens,
        )

    def _request_ollama_model(
        self,
        model: str,
        messages: list,
        system_prompt: str,
        max_tokens: int,
    ) -> dict:
        model_messages = [
            {
                "role": "system",
                "content": system_prompt,
            }
        ]

        model_messages.extend(messages)

        settings = MODEL_SETTINGS.get(
            model,
            {
                "num_ctx": DEFAULT_CONTEXT,
                "num_predict": max_tokens,
                "temperature": 0.3,
            },
        ).copy()

        settings["num_predict"] = max_tokens

        payload = {
            "model": model,
            "messages": model_messages,
            "stream": False,
            "options": settings,
        }

        started = time.time()

        try:
            response = requests.post(
                self.chat_url,
                json=payload,
                timeout=self.timeout,
            )

            elapsed = round(time.time() - started, 2)

            if not response.ok:
                try:
                    error_data = response.json()
                except Exception:
                    error_data = response.text

                error_message = (
                    error_data.get("error", str(error_data))
                    if isinstance(error_data, dict)
                    else str(error_data)
                )

                self.last_errors[model] = error_message

                return {
                    "success": False,
                    "complete": False,
                    "model": model,
                    "text": "",
                    "error": error_message,
                    "status_code": response.status_code,
                    "elapsed": elapsed,
                }

            data = response.json()
            message_data = data.get("message", {})
            output_text = message_data.get("content", "")

            complete = (
                bool(output_text.strip())
                and not data.get("done_reason") == "length"
                and data.get("done", True)
            )

            usage = {
                "prompt_eval_count": data.get("prompt_eval_count"),
                "eval_count": data.get("eval_count"),
                "total_duration": data.get("total_duration"),
                "load_duration": data.get("load_duration"),
            }

            self.last_usage[model] = usage

            return {
                "success": bool(output_text.strip()),
                "complete": complete,
                "model": model,
                "text": output_text.strip(),
                "error": None if output_text.strip() else (
                    "Ollama вернула пустой ответ."
                ),
                "elapsed": elapsed,
                "usage": usage,
                "done_reason": data.get("done_reason"),
            }

        except Exception as exc:
            self.last_errors[model] = str(exc)

            return {
                "success": False,
                "complete": False,
                "model": model,
                "text": "",
                "error": str(exc),
            }

    # ---------------------------------------------------------
    # GENERATION WITH RETRY
    # ---------------------------------------------------------

    def ask_model(
        self,
        model: str,
        prompt: str,
        context=None,
        system_prompt: Optional[str] = None,
        max_tokens: Optional[int] = None,
    ) -> dict:
        task_type = self.classify_task(prompt)

        if system_prompt is None:
            system_prompt = self.build_system_prompt(task_type)

        messages = self.clean_context(context)
        messages.append(
            {
                "role": "user",
                "content": prompt,
            }
        )

        token_limit = max_tokens or DEFAULT_NUM_PREDICT

        result = self._request_model(
            model,
            messages,
            system_prompt,
            token_limit,
        )

        if (
            result.get("success")
            and not result.get("complete")
            and token_limit < RETRY_NUM_PREDICT
        ):
            retry = self._request_model(
                model,
                messages,
                system_prompt,
                RETRY_NUM_PREDICT,
            )

            if retry.get("success"):
                retry["retried"] = True
                return retry

        return result

    # ---------------------------------------------------------
    # COMPARISON
    # ---------------------------------------------------------

    def compare_answers(
        self,
        prompt: str,
        answers: list,
        comparison_model: Optional[str] = None,
    ) -> dict:
        if not answers:
            return {
                "success": False,
                "text": "",
                "error": "Нет ответов для сравнения.",
            }

        comparison_model = (
            comparison_model or DEFAULT_COMPARISON_MODEL
        )

        formatted_answers = []

        for index, answer in enumerate(answers, start=1):
            if not isinstance(answer, dict):
                continue

            formatted_answers.append(
                f"Ответ {index} — модель {answer.get('model', 'unknown')}:\n"
                f"{answer.get('text', '')}"
            )

        comparison_prompt = f"""
Сравни ответы моделей на запрос пользователя.

Запрос пользователя:
{prompt}

Ответы моделей:
{chr(10).join(formatted_answers)}

Проведи критический анализ:
1. Какие утверждения совпадают?
2. Какие утверждения противоречат друг другу?
3. Какие ответы содержат конкретные аргументы или проверяемые детали?
4. Какие части требуют дополнительной проверки?
5. Какой ответ лучше подходит к запросу и почему?

Не считай совпадение нескольких моделей доказательством истинности.
Не выдумывай источники и результаты проверок.
В конце дай объединённый ответ, отдельно обозначив неопределённость.
""".strip()

        return self.ask_model(
            comparison_model,
            comparison_prompt,
            system_prompt=self.build_system_prompt("complex"),
            max_tokens=COMPARISON_NUM_PREDICT,
        )

    # ---------------------------------------------------------
    # RESULT VERIFICATION
    # ---------------------------------------------------------

    def verify_result(
        self,
        result,
        verification_callback: Optional[Callable] = None,
    ) -> dict:
        if not isinstance(result, dict):
            return {
                "verified": False,
                "evidence": [],
                "reason": "Результат имеет неверный формат.",
            }

        if not result.get("success"):
            return {
                "verified": False,
                "evidence": [],
                "reason": "Модель не вернула успешный результат.",
            }

        if verification_callback is None:
            return {
                "verified": False,
                "evidence": [],
                "reason": (
                    "Проверка не выполнена: отсутствует "
                    "verification_callback."
                ),
            }

        try:
            verification = verification_callback(result)

            if isinstance(verification, dict):
                evidence = verification.get("evidence", [])
                verified = (
                    bool(verification.get("verified"))
                    and isinstance(evidence, list)
                    and bool(evidence)
                )

                return {
                    "verified": verified,
                    "evidence": evidence if isinstance(evidence, list) else [],
                    "reason": verification.get("reason"),
                }

            if isinstance(verification, bool):
                return {
                    "verified": False,
                    "evidence": [],
                    "reason": (
                        "Проверка вернула только логический результат, "
                        "без подтверждающих данных."
                    ),
                }

            return {
                "verified": False,
                "evidence": [],
                "reason": "Неизвестный формат результата проверки.",
            }

        except Exception as exc:
            return {
                "verified": False,
                "evidence": [],
                "reason": str(exc),
            }

    # ---------------------------------------------------------
    # COORDINATION
    # ---------------------------------------------------------

    def coordinate(
        self,
        prompt: str,
        context=None,
        compare: bool = False,
        verify: bool = False,
        verification_callback: Optional[Callable] = None,
        max_models: int = 3,
    ) -> dict:
        started = time.time()

        task_type = self.classify_task(prompt)
        available = self.get_available_model_names()

        preferred = self.select_models(task_type)

        if not preferred:
            return {
                "success": False,
                "complete": False,
                "verified": False,
                "task_type": task_type,
                "models_used": [],
                "responses": [],
                "answer": "",
                "comparison": None,
                "verification": {
                    "verified": False,
                    "evidence": [],
                    "reason": (
                        "Нет доступных моделей. Проверь Ollama "
                        "и настройки ключей в KeyManager."
                    ),
                },
                "error": "Нет доступных моделей.",
                "elapsed": round(time.time() - started, 2),
            }

        preferred = preferred[:max(1, max_models)]

        system_prompt = self.build_system_prompt(task_type)
        responses = []
        successful = []

        for model in preferred:
            result = self.ask_model(
                model=model,
                prompt=prompt,
                context=context,
                system_prompt=system_prompt,
            )

            responses.append(result)

            if result.get("success"):
                successful.append(result)

            if not compare and result.get("success") and result.get("complete"):
                break

        comparison = None

        if compare and successful:
            comparison_model = DEFAULT_COMPARISON_MODEL

            if comparison_model not in available:
                comparison_model = successful[0]["model"]

            comparison = self.compare_answers(
                prompt,
                successful,
                comparison_model,
            )

        if comparison and comparison.get("success"):
            answer = comparison.get("text", "")
            final_model = comparison.get("model")
            complete = comparison.get("complete", False)
        elif successful:
            best = next(
                (
                    item
                    for item in successful
                    if item.get("complete")
                ),
                successful[0],
            )

            answer = best.get("text", "")
            final_model = best.get("model")
            complete = best.get("complete", False)
        else:
            answer = ""
            final_model = None
            complete = False

        verification_result = {
            "verified": False,
            "evidence": [],
            "reason": "Проверка не запрашивалась.",
        }

        if verify:
            source_result = {
                "success": bool(answer),
                "complete": complete,
                "model": final_model,
                "text": answer,
            }

            verification_result = self.verify_result(
                source_result,
                verification_callback,
            )

        return {
            "success": bool(answer),
            "complete": complete,
            "verified": verification_result.get("verified", False),
            "task_type": task_type,
            "available_models": available,
            "models_used": [
                item.get("model")
                for item in responses
                if item.get("success")
            ],
            "responses": responses,
            "answer": answer,
            "final_model": final_model,
            "comparison": comparison,
            "verification": verification_result,
            "errors": {
                **self.last_errors,
            },
            "usage": {
                **self.last_usage,
            },
            "elapsed": round(time.time() - started, 2),
            "timestamp": datetime.now().isoformat(timespec="seconds"),
        }

    def status(self) -> dict:
        ollama = self.list_models()

        provider_status = {}

        providers = (
            "openai",
            "openrouter",
            "groq",
            "gemini",
            "anthropic",
            "mistral",
            "deepseek",
            "cohere",
            "huggingface",
            "cloudflare",
            "nvidia",
            "replicate",
            "together",
            "fireworks",
            "perplexity",
        )

        for provider in providers:
            provider_status[provider] = {
                "has_key": self._provider_has_key(provider),
            }

        try:
            credentials = self.key_manager.list_credentials()

            if isinstance(credentials, dict):
                for name, data in credentials.items():
                    if name in provider_status and isinstance(data, dict):
                        provider_status[name]["enabled"] = data.get(
                            "enabled",
                            True,
                        )
        except Exception:
            pass

        return {
            "ollama": ollama,
            "providers": provider_status,
            "available_models": self.get_available_model_names(),
            "timestamp": datetime.now().isoformat(timespec="seconds"),
        }

    def test_classifier(self) -> list:
        tests = [
            ("Привет, как дела?", "simple"),
            ("Проанализируй архитектуру AI Core", "complex"),
            ("Напиши Python функцию", "code"),
            ("Проанализируй изображение", "vision"),
            ("Подключи микрофон и распознавание речи", "audio"),
        ]

        results = []

        for text, expected in tests:
            actual = self.classify_task(text)

            results.append(
                {
                    "input": text,
                    "expected": expected,
                    "actual": actual,
                    "passed": expected == actual,
                }
            )

        return results

_default_coordinator = None


def get_coordinator() -> ModelCoordinator:
    global _default_coordinator

    if _default_coordinator is None:
        _default_coordinator = ModelCoordinator()

    return _default_coordinator

    context=None,
    compare: bool = False,
    verify: bool = False,
    verification_callback: Optional[Callable] = None,

    prompt: str,
